## 2. 数字序列匹配算法

### 2.1 算法原理

从卡片名称和文件名称中提取数字序列，建立数字到卡片/文件的映射，然后通过数字匹配建立绑定关系。

### 2.2 详细步骤

**步骤1：数字提取**

```typescript
// 提取字符串中的所有数字序列
function extractNumbers(text: string): NumberInfo[] {
  const regex = /\d+/g;
  const results: NumberInfo[] = [];
  let match: RegExpExecArray | null;
  
  while ((match = regex.exec(text)) !== null) {
    results.push({
      value: parseInt(match[0], 10),
      raw: match[0],
      position: match.index,
      length: match[0].length
    });
  }
  
  return results;
}

// 示例
extractNumbers("Season1_EP01_HD")
// → [
//   { value: 1, raw: "1", position: 6, length: 1 },
//   { value: 1, raw: "01", position: 10, length: 2 }
// ]
```

**步骤2：数字标准化**

```typescript
// 数字标准化（处理前导零）
function normalizeNumber(raw: string): number {
  return parseInt(raw, 10);
}

// "01" → 1
// "001" → 1
// "10" → 10
```

**步骤3：选择匹配数字**

当名称中有多个数字时，需要决定使用哪个数字进行匹配：

```typescript
type NumberPosition = 'first' | 'last' | 'longest' | 'auto';

function selectNumber(
  numbers: NumberInfo[],
  position: NumberPosition
): NumberInfo | null {
  if (numbers.length === 0) return null;
  
  switch (position) {
    case 'first':
      return numbers[0];
    case 'last':
      return numbers[numbers.length - 1];
    case 'longest':
      return numbers.reduce((a, b) => 
        a.length > b.length ? a : b
      );
    case 'auto':
      // 智能选择：优先选择2-3位数字
      const preferred = numbers.find(n => 
        n.length >= 2 && n.length <= 3
      );
      return preferred || numbers[numbers.length - 1];
  }
}
```

**步骤4：建立匹配**

```typescript
function buildNumberMatches(
  cards: CardInfo[],
  files: FileInfo[],
  config: NumberMatchConfig
): MatchResult {
  // 建立卡片数字映射
  const cardMap = new Map<number, CardInfo[]>();
  for (const card of cards) {
    const numbers = extractNumbers(card.name);
    const selected = selectNumber(numbers, config.cardPosition);
    if (selected) {
      const key = selected.value;
      if (!cardMap.has(key)) cardMap.set(key, []);
      cardMap.get(key)!.push(card);
    }
  }
  
  // 建立文件数字映射
  const fileMap = new Map<number, FileInfo[]>();
  for (const file of files) {
    const numbers = extractNumbers(file.name);
    const selected = selectNumber(numbers, config.filePosition);
    if (selected) {
      const key = selected.value;
      if (!fileMap.has(key)) fileMap.set(key, []);
      fileMap.get(key)!.push(file);
    }
  }
  
  // 匹配
  const bindings: BindingItem[] = [];
  const conflicts: ConflictInfo[] = [];
  
  for (const [number, cardList] of cardMap) {
    const fileList = fileMap.get(number);
    if (fileList && fileList.length > 0) {
      if (cardList.length === 1 && fileList.length === 1) {
        // 一对一匹配
        bindings.push({
          cardId: cardList[0].id,
          filePath: fileList[0].path,
          resourceField: detectResourceField(cardList[0], fileList[0])
        });
      } else {
        // 冲突：多个卡片或多个文件有相同数字
        conflicts.push({
          number,
          cards: cardList.map(c => c.id),
          files: fileList.map(f => f.path)
        });
      }
    }
  }
  
  return { bindings, conflicts };
}
```

### 2.3 冲突处理

当多个卡片或文件有相同数字时：

```typescript
interface ConflictInfo {
  number: number;
  cards: string[];
  files: string[];
}

function resolveConflicts(
  conflicts: ConflictInfo[],
  strategy: 'first' | 'manual' | 'skip'
): BindingItem[] {
  const resolved: BindingItem[] = [];
  
  for (const conflict of conflicts) {
    switch (strategy) {
      case 'first':
        // 使用第一个
        if (conflict.cards.length > 0 && conflict.files.length > 0) {
          resolved.push({
            cardId: conflict.cards[0],
            filePath: conflict.files[0],
            resourceField: 'primary_resource'
          });
        }
        break;
      case 'skip':
        // 跳过冲突项
        break;
      case 'manual':
        // 标记为需要手动处理
        break;
    }
  }
  
  return resolved;
}
```

### 2.4 置信度计算

```typescript
function calculateNumberMatchConfidence(
  result: MatchResult,
  totalCards: number,
  totalFiles: number
): number {
  const matchedCount = result.bindings.length;
  const conflictCount = result.conflicts.length;
  
  // 基础分数：匹配率
  let score = (matchedCount / Math.max(totalCards, totalFiles)) * 100;
  
  // 惩罚冲突
  score -= conflictCount * 5;
  
  // 惩罚未匹配
  const unmatchedCards = totalCards - matchedCount - conflictCount;
  const unmatchedFiles = totalFiles - matchedCount - conflictCount;
  score -= (unmatchedCards + unmatchedFiles) * 2;
  
  return Math.max(0, Math.min(100, score));
}
```

## 3. 关键词匹配算法

### 3.1 算法原理

从名称中提取关键词，计算卡片和文件之间的关键词相似度，选择最佳匹配。

### 3.2 分词实现

```typescript
// 多语言分词
class Tokenizer {
  // 中文分词（简单实现：按字分割）
  tokenizeChinese(text: string): string[] {
    return text.match(/[\u4e00-\u9fa5]/g) || [];
  }
  
  // 英文分词
  tokenizeEnglish(text: string): string[] {
    return text
      .toLowerCase()
      .split(/[^a-z0-9]+/)
      .filter(t => t.length > 1);
  }
  
  // 日文分词（简单实现）
  tokenizeJapanese(text: string): string[] {
    // 平假名、片假名、汉字
    return text.match(/[\u3040-\u309f\u30a0-\u30ff\u4e00-\u9fa5]+/g) || [];
  }
  
  // 综合分词
  tokenize(text: string): string[] {
    const tokens: string[] = [];
    
    // 提取中文
    tokens.push(...this.tokenizeChinese(text));
    
    // 提取英文
    tokens.push(...this.tokenizeEnglish(text));
    
    // 提取数字
    tokens.push(...(text.match(/\d+/g) || []));
    
    // 去重
    return [...new Set(tokens)];
  }
}
```

### 3.3 相似度计算

**Jaccard相似度**：

```typescript
function jaccardSimilarity(set1: Set<string>, set2: Set<string>): number {
  if (set1.size === 0 && set2.size === 0) return 0;
  
  const intersection = new Set([...set1].filter(x => set2.has(x)));
  const union = new Set([...set1, ...set2]);
  
  return intersection.size / union.size;
}
```

**余弦相似度**：

```typescript
function cosineSimilarity(vec1: number[], vec2: number[]): number {
  let dotProduct = 0;
  let norm1 = 0;
  let norm2 = 0;
  
  for (let i = 0; i < vec1.length; i++) {
    dotProduct += vec1[i] * vec2[i];
    norm1 += vec1[i] * vec1[i];
    norm2 += vec2[i] * vec2[i];
  }
  
  if (norm1 === 0 || norm2 === 0) return 0;
  return dotProduct / (Math.sqrt(norm1) * Math.sqrt(norm2));
}
```

**编辑距离相似度**：

```typescript
function editDistanceSimilarity(s1: string, s2: string): number {
  const distance = levenshteinDistance(s1, s2);
  const maxLen = Math.max(s1.length, s2.length);
  if (maxLen === 0) return 1;
  return 1 - distance / maxLen;
}

function levenshteinDistance(s1: string, s2: string): number {
  const m = s1.length;
  const n = s2.length;
  const dp: number[][] = Array(m + 1)
    .fill(null)
    .map(() => Array(n + 1).fill(0));
  
  for (let i = 0; i <= m; i++) dp[i][0] = i;
  for (let j = 0; j <= n; j++) dp[0][j] = j;
  
  for (let i = 1; i <= m; i++) {
    for (let j = 1; j <= n; j++) {
      if (s1[i - 1] === s2[j - 1]) {
        dp[i][j] = dp[i - 1][j - 1];
      } else {
        dp[i][j] = Math.min(
          dp[i - 1][j] + 1,
          dp[i][j - 1] + 1,
          dp[i - 1][j - 1] + 1
        );
      }
    }
  }
  
  return dp[m][n];
}
```

### 3.4 综合匹配

```typescript
class KeywordMatcher {
  private tokenizer = new Tokenizer();
  private config: KeywordMatchConfig;
  
  constructor(config: KeywordMatchConfig = {}) {
    this.config = {
      minSimilarity: 0.3,
      weightJaccard: 0.5,
      weightEdit: 0.3,
      weightNumber: 0.2,
      ...config
    };
  }
  
  // 计算综合相似度
  calculateSimilarity(cardName: string, fileName: string): number {
    const cardTokens = new Set(this.tokenizer.tokenize(cardName));
    const fileTokens = new Set(this.tokenizer.tokenize(fileName));
    
    // Jaccard相似度
    const jaccard = jaccardSimilarity(cardTokens, fileTokens);
    
    // 编辑距离相似度
    const edit = editDistanceSimilarity(
      cardName.toLowerCase(),
      fileName.toLowerCase()
    );
    
    // 数字匹配加分
    const cardNumbers = cardName.match(/\d+/g) || [];
    const fileNumbers = fileName.match(/\d+/g) || [];
    const numberMatch = cardNumbers.some(cn =>
      fileNumbers.some(fn => parseInt(cn) === parseInt(fn))
    ) ? 1 : 0;
    
    // 加权平均
    return (
      jaccard * this.config.weightJaccard! +
      edit * this.config.weightEdit! +
      numberMatch * this.config.weightNumber!
    );
  }
  
  // 执行匹配
  match(cards: CardInfo[], files: FileInfo[]): MatchScheme {
    // 计算相似度矩阵
    const similarities: SimilarityEntry[] = [];
    
    for (const card of cards) {
      for (const file of files) {
        const score = this.calculateSimilarity(card.name, file.name);
        if (score >= this.config.minSimilarity!) {
          similarities.push({ cardId: card.id, filePath: file.path, score });
        }
      }
    }
    
    // 按分数排序
    similarities.sort((a, b) => b.score - a.score);
    
    // 贪心选择
    const usedCards = new Set<string>();
    const usedFiles = new Set<string>();
    const bindings: BindingItem[] = [];
    
    for (const { cardId, filePath } of similarities) {
      if (!usedCards.has(cardId) && !usedFiles.has(filePath)) {
        bindings.push({ cardId, filePath, resourceField: 'primary_resource' });
        usedCards.add(cardId);
        usedFiles.add(filePath);
      }
    }
    
    return {
      mode: 'keyword',
      confidence: this.calculateConfidence(bindings.length, cards.length),
      bindings,
      unmatchedCards: cards.filter(c => !usedCards.has(c.id)).map(c => c.id),
      unmatchedFiles: files.filter(f => !usedFiles.has(f.path)).map(f => f.path)
    };
  }
}
```

## 4. 文件顺序匹配算法

### 4.1 算法原理

最简单的匹配方式：将卡片和文件分别排序，然后按顺序一一对应。

### 4.2 排序策略

```typescript
type SortStrategy = 'name' | 'name_numeric' | 'created' | 'modified' | 'size';

class OrderMatcher {
  // 按名称排序（字典序）
  sortByName<T extends { name: string }>(items: T[]): T[] {
    return [...items].sort((a, b) => a.name.localeCompare(b.name));
  }
  
  // 按名称排序（数字优先）
  sortByNameNumeric<T extends { name: string }>(items: T[]): T[] {
    return [...items].sort((a, b) => {
      // 提取数字
      const numA = parseInt(a.name.match(/\d+/)?.[0] || '0');
      const numB = parseInt(b.name.match(/\d+/)?.[0] || '0');
      
      if (numA !== numB) {
        return numA - numB;
      }
      return a.name.localeCompare(b.name);
    });
  }
  
  // 按时间排序
  sortByTime<T extends { time: string }>(
    items: T[],
    order: 'asc' | 'desc' = 'asc'
  ): T[] {
    return [...items].sort((a, b) => {
      const diff = new Date(a.time).getTime() - new Date(b.time).getTime();
      return order === 'asc' ? diff : -diff;
    });
  }
}
```

### 4.3 匹配实现

```typescript
class FileOrderMatcher {
  match(
    cards: CardInfo[],
    files: FileInfo[],
    options: OrderMatchOptions = {}
  ): MatchScheme {
    const {
      cardSort = 'name_numeric',
      fileSort = 'name_numeric',
      fillMode = 'truncate'  // 'truncate' | 'repeat' | 'skip'
    } = options;
    
    // 排序
    const sortedCards = this.sort(cards, cardSort, 'name');
    const sortedFiles = this.sort(files, fileSort, 'name');
    
    // 匹配
    const bindings: BindingItem[] = [];
    const minLen = Math.min(sortedCards.length, sortedFiles.length);
    
    for (let i = 0; i < minLen; i++) {
      bindings.push({
        cardId: sortedCards[i].id,
        filePath: sortedFiles[i].path,
        resourceField: 'primary_resource'
      });
    }
    
    // 计算置信度
    const countMatch = sortedCards.length === sortedFiles.length;
    const confidence = countMatch ? 85 : Math.max(50, 85 - Math.abs(sortedCards.length - sortedFiles.length) * 5);
    
    return {
      mode: 'file_order',
      confidence,
      bindings,
      unmatchedCards: sortedCards.slice(minLen).map(c => c.id),
      unmatchedFiles: sortedFiles.slice(minLen).map(f => f.path),
      warning: !countMatch 
        ? `数量不匹配：${sortedCards.length}个卡片，${sortedFiles.length}个文件`
        : undefined
    };
  }
  
  private sort<T>(
    items: T[],
    strategy: SortStrategy,
    nameKey: keyof T
  ): T[] {
    switch (strategy) {
      case 'name':
        return [...items].sort((a, b) => 
          String(a[nameKey]).localeCompare(String(b[nameKey]))
        );
      case 'name_numeric':
        return [...items].sort((a, b) => {
          const numA = parseInt(String(a[nameKey]).match(/\d+/)?.[0] || '0');
          const numB = parseInt(String(b[nameKey]).match(/\d+/)?.[0] || '0');
          return numA - numB || 
            String(a[nameKey]).localeCompare(String(b[nameKey]));
        });
      default:
        return [...items];
    }
  }
}
```

## 5. 方案综合评估

### 5.1 多方案生成

```typescript
class SchemeGenerator {
  private numberMatcher = new NumberSequenceMatcher();
  private keywordMatcher = new KeywordMatcher();
  private orderMatcher = new FileOrderMatcher();
  
  generateAll(cards: CardInfo[], files: FileInfo[]): MatchScheme[] {
    const schemes: MatchScheme[] = [];
    
    // 数字序列匹配
    const numberScheme = this.numberMatcher.match(cards, files);
    if (numberScheme.bindings.length > 0) {
      schemes.push(numberScheme);
    }
    
    // 关键词匹配
    const keywordScheme = this.keywordMatcher.match(cards, files);
    if (keywordScheme.bindings.length > 0) {
      schemes.push(keywordScheme);
    }
    
    // 文件顺序匹配
    const orderScheme = this.orderMatcher.match(cards, files);
    schemes.push(orderScheme);
    
    // 按置信度排序
    return schemes.sort((a, b) => b.confidence - a.confidence);
  }
}
```

### 5.2 方案评分

```typescript
interface SchemeScore {
  scheme: MatchScheme;
  scores: {
    matchRate: number;      // 匹配率
    conflictRate: number;   // 冲突率
    coverage: number;       // 覆盖率
    consistency: number;    // 一致性
  };
  totalScore: number;
}

function evaluateScheme(
  scheme: MatchScheme,
  totalCards: number,
  totalFiles: number
): SchemeScore {
  // 匹配率
  const matchRate = scheme.bindings.length / Math.max(totalCards, totalFiles);
  
  // 冲突率（越低越好）
  const conflictRate = 0; // 假设冲突已解决
  
  // 覆盖率
  const coverage = (
    (totalCards - scheme.unmatchedCards.length) / totalCards +
    (totalFiles - scheme.unmatchedFiles.length) / totalFiles
  ) / 2;
  
  // 一致性（卡片和文件数量是否一致）
  const consistency = totalCards === totalFiles ? 1 : 
    1 - Math.abs(totalCards - totalFiles) / Math.max(totalCards, totalFiles);
  
  // 综合评分
  const totalScore = (
    matchRate * 0.4 +
    (1 - conflictRate) * 0.2 +
    coverage * 0.3 +
    consistency * 0.1
  ) * 100;
  
  return {
    scheme,
    scores: { matchRate, conflictRate, coverage, consistency },
    totalScore
  };
}
```
