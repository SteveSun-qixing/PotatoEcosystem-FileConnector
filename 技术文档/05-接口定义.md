## 1. 核心类接口

### 1.1 FileConnector 主类

```typescript
/**
 * 文件连接器主类
 * 提供文件与卡片批量绑定的核心功能
 */
export class FileConnector extends EventEmitter {
  /**
   * 创建文件连接器实例
   * @param options 配置选项
   */
  constructor(options: FileConnectorOptions);
  
  /**
   * 初始化连接器
   * 加载配置、建立SDK连接
   */
  async initialize(): Promise<void>;
  
  /**
   * 加载卡片列表
   * @param source 卡片来源（箱子路径或卡片数组）
   */
  async loadCards(source: string | CardInfo[]): Promise<CardInfo[]>;
  
  /**
   * 加载文件列表
   * @param paths 文件路径或文件夹路径
   * @param options 加载选项
   */
  async loadFiles(
    paths: string[],
    options?: LoadFilesOptions
  ): Promise<FileInfo[]>;
  
  /**
   * 手动设置绑定
   * @param bindings 绑定列表
   */
  setBindings(bindings: BindingItem[]): void;
  
  /**
   * 执行自动匹配
   * @param mode 匹配模式
   * @returns 匹配方案列表
   */
  async autoMatch(mode?: MatchMode): Promise<MatchScheme[]>;
  
  /**
   * 执行智能匹配
   * @returns 智能匹配结果
   */
  async smartMatch(): Promise<SmartMatchResult>;
  
  /**
   * 应用匹配方案
   * @param scheme 要应用的方案
   */
  applyScheme(scheme: MatchScheme): void;
  
  /**
   * 执行绑定
   * @param options 执行选项
   * @returns 执行结果
   */
  async execute(options?: ExecuteOptions): Promise<ExecuteResult>;
  
  /**
   * 批量转换资源模式
   * @param cardIds 卡片ID列表
   * @param targetMode 目标模式
   * @param options 转换选项
   */
  async batchConvert(
    cardIds: string[],
    targetMode: 'full' | 'empty',
    options?: ConvertOptions
  ): Promise<ConvertResult>;
  
  /**
   * 获取当前状态
   */
  getState(): ConnectorState;
  
  /**
   * 重置连接器状态
   */
  reset(): void;
  
  /**
   * 销毁连接器
   */
  destroy(): void;
}
```

### 1.2 事件接口

```typescript
/**
 * 文件连接器事件
 */
interface FileConnectorEvents {
  // 卡片加载
  'cards:loading': void;
  'cards:loaded': { cards: CardInfo[] };
  'cards:error': { error: Error };
  
  // 文件加载
  'files:loading': void;
  'files:loaded': { files: FileInfo[] };
  'files:error': { error: Error };
  
  // 匹配
  'match:start': { mode: MatchMode };
  'match:complete': { schemes: MatchScheme[] };
  'match:error': { error: Error };
  
  // 绑定变化
  'bindings:change': { bindings: BindingItem[] };
  
  // 执行
  'execute:start': { total: number };
  'execute:progress': ExecutionProgress;
  'execute:item:success': { binding: BindingItem };
  'execute:item:error': { binding: BindingItem; error: Error };
  'execute:complete': { result: ExecuteResult };
  
  // 转换
  'convert:start': { cardIds: string[]; targetMode: string };
  'convert:progress': ConvertProgress;
  'convert:complete': { result: ConvertResult };
}
```

## 2. 匹配器接口

### 2.1 基础匹配器接口

```typescript
/**
 * 匹配器基础接口
 */
interface IMatcher {
  /** 匹配器名称 */
  readonly name: string;
  
  /** 匹配模式 */
  readonly mode: MatchMode;
  
  /**
   * 执行匹配
   * @param cards 卡片列表
   * @param files 文件列表
   * @param options 匹配选项
   */
  match(
    cards: CardInfo[],
    files: FileInfo[],
    options?: MatchOptions
  ): MatchScheme;
  
  /**
   * 检查是否适用
   * @param cards 卡片列表
   * @param files 文件列表
   */
  isApplicable(cards: CardInfo[], files: FileInfo[]): boolean;
}
```

### 2.2 数字匹配器

```typescript
/**
 * 数字序列匹配器

export class NumberSequenceMatcher implements IMatcher {

  constructor(config?: NumberMatchConfig);

   * 提取数字

  /**
   * 选择用于匹配的数字
   */

   * 执行匹配
   */
  match(cards: CardInfo[], files: FileInfo[], options?: MatchOptions): MatchScheme;
  
  /**
   * 检查是否适用（检测是否有数字规律）
   */
  isApplicable(cards: CardInfo[], files: FileInfo[]): boolean;
  
  /**
   * 解决冲突
   */
  resolveConflicts(conflicts: MatchConflict[], strategy: ConflictStrategy): BindingItem[];
}
```
