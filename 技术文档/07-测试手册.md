```typescript
// tests/unit/matchers/NumberSequenceMatcher.test.ts

import { describe, it, expect, beforeEach } from 'vitest';
import { NumberSequenceMatcher } from '@/core/matchers';
import type { CardInfo, FileInfo } from '@/types';

describe('NumberSequenceMatcher', () => {
  let matcher: NumberSequenceMatcher;
  
  beforeEach(() => {
    matcher = new NumberSequenceMatcher();
  });
  
  describe('extractNumbers', () => {
    it('should extract single number', () => {
      const result = matcher.extractNumbers('第01集');
      expect(result).toHaveLength(1);
      expect(result[0].value).toBe(1);
    });
    
    it('should extract multiple numbers', () => {
      const result = matcher.extractNumbers('Season1_EP02');
      expect(result).toHaveLength(2);
      expect(result[0].value).toBe(1);
      expect(result[1].value).toBe(2);
    });
    
    it('should return empty array for no numbers', () => {
      const result = matcher.extractNumbers('no numbers here');
      expect(result).toHaveLength(0);
    });
    
    it('should handle leading zeros', () => {
      const result = matcher.extractNumbers('EP001');
      expect(result[0].value).toBe(1);
      expect(result[0].raw).toBe('001');
    });
  });
  
  describe('match', () => {
    it('should match cards and files by number', () => {
      const cards: CardInfo[] = [
        { id: '1', name: '第01集', type: 'video', path: '', baseCards: [], resourceMode: 'empty', metadata: {} as any },
        { id: '2', name: '第02集', type: 'video', path: '', baseCards: [], resourceMode: 'empty', metadata: {} as any },
      ];
      
      const files: FileInfo[] = [
        { name: 'EP01.mp4', path: '/EP01.mp4', size: 100, mimeType: 'video/mp4', type: 'video', extension: 'mp4', modifiedAt: '', createdAt: '' },
        { name: 'EP02.mp4', path: '/EP02.mp4', size: 100, mimeType: 'video/mp4', type: 'video', extension: 'mp4', modifiedAt: '', createdAt: '' },
      ];
      
      const result = matcher.match(cards, files);
      
      expect(result.bindings).toHaveLength(2);
      expect(result.bindings[0].cardId).toBe('1');
      expect(result.bindings[0].filePath).toBe('/EP01.mp4');
      expect(result.confidence).toBeGreaterThan(80);
    });
    
    it('should handle unmatched items', () => {
      const cards: CardInfo[] = [
        { id: '1', name: '第01集', type: 'video', path: '', baseCards: [], resourceMode: 'empty', metadata: {} as any },
        { id: '2', name: '第03集', type: 'video', path: '', baseCards: [], resourceMode: 'empty', metadata: {} as any },
      ];
      
      const files: FileInfo[] = [
        { name: 'EP01.mp4', path: '/EP01.mp4', size: 100, mimeType: 'video/mp4', type: 'video', extension: 'mp4', modifiedAt: '', createdAt: '' },
        { name: 'EP02.mp4', path: '/EP02.mp4', size: 100, mimeType: 'video/mp4', type: 'video', extension: 'mp4', modifiedAt: '', createdAt: '' },
      ];
      
      const result = matcher.match(cards, files);
      
      expect(result.bindings).toHaveLength(1);
      expect(result.unmatchedCards).toContain('2');
      expect(result.unmatchedFiles).toContain('/EP02.mp4');
    });
  });
});
```

```typescript
// tests/unit/matchers/KeywordMatcher.test.ts

import { describe, it, expect, beforeEach } from 'vitest';
import { KeywordMatcher } from '@/core/matchers';

describe('KeywordMatcher', () => {
  let matcher: KeywordMatcher;
  
  beforeEach(() => {
    matcher = new KeywordMatcher();
  });
  
  describe('tokenize', () => {
    it('should tokenize English text', () => {
      const tokens = matcher.tokenize('Hello World');
      expect(tokens).toContain('hello');
      expect(tokens).toContain('world');
    });
    
    it('should tokenize Chinese text', () => {
      const tokens = matcher.tokenize('孤独摇滚');
      expect(tokens.length).toBeGreaterThan(0);
    });
    
    it('should tokenize mixed text', () => {
      const tokens = matcher.tokenize('孤独摇滚 EP01');
      expect(tokens.length).toBeGreaterThan(2);
    });
  });
  
  describe('calculateSimilarity', () => {
    it('should return high similarity for similar names', () => {
      const similarity = matcher.calculateSimilarity(
        '孤独摇滚 第一集',
        '孤独摇滚 01'
      );
      expect(similarity).toBeGreaterThan(0.5);
    });
    
    it('should return low similarity for different names', () => {
      const similarity = matcher.calculateSimilarity(
        '进击的巨人',
        'Pokemon Episode'
      );
      expect(similarity).toBeLessThan(0.3);
    });
  });
});
```

### 2.2 执行器测试

**BindingExecutor测试**

```typescript
// tests/unit/executors/BindingExecutor.test.ts

import { describe, it, expect, vi, beforeEach } from 'vitest';
import { BindingExecutor } from '@/core/executors';
import type { BindingItem } from '@/types';

describe('BindingExecutor', () => {
  let executor: BindingExecutor;
  let mockSDK: any;
  
  beforeEach(() => {
    mockSDK = {
      card: {
        get: vi.fn(),
        update: vi.fn()
      },
      resources: {
        exists: vi.fn()
      }
    };
    executor = new BindingExecutor(mockSDK);
  });
  
  describe('execute', () => {
    it('should execute all bindings successfully', async () => {
      mockSDK.resources.exists.mockResolvedValue(true);
      mockSDK.card.update.mockResolvedValue({ success: true });
      
      const bindings: BindingItem[] = [
        { cardId: '1', filePath: '/file1.mp4', resourceField: 'video_file' },
        { cardId: '2', filePath: '/file2.mp4', resourceField: 'video_file' }
      ];
      
      const result = await executor.execute(bindings);
      
      expect(result.success).toBe(2);
      expect(result.failed).toBe(0);
    });
    
    it('should handle errors gracefully', async () => {
      mockSDK.resources.exists.mockResolvedValue(true);
      mockSDK.card.update
        .mockResolvedValueOnce({ success: true })
        .mockRejectedValueOnce(new Error('Update failed'));
      
      const bindings: BindingItem[] = [
        { cardId: '1', filePath: '/file1.mp4', resourceField: 'video_file' },
        { cardId: '2', filePath: '/file2.mp4', resourceField: 'video_file' }
      ];
      
      const result = await executor.execute(bindings, { errorStrategy: 'skip' });
      
      expect(result.success).toBe(1);
      expect(result.failed).toBe(1);
      expect(result.errors).toHaveLength(1);
    });
    
    it('should call progress callback', async () => {
      mockSDK.resources.exists.mockResolvedValue(true);
      mockSDK.card.update.mockResolvedValue({ success: true });
      
      const progressCallback = vi.fn();
      const bindings: BindingItem[] = [
        { cardId: '1', filePath: '/file1.mp4', resourceField: 'video_file' }
      ];
      
      await executor.execute(bindings, { onProgress: progressCallback });
      
      expect(progressCallback).toHaveBeenCalled();
    });
  });
  
  describe('validateBinding', () => {
    it('should validate binding with existing file', async () => {
      mockSDK.resources.exists.mockResolvedValue(true);
      
      const binding: BindingItem = {
        cardId: '1',
        filePath: '/existing.mp4',
        resourceField: 'video_file'
      };
      
      const result = await executor.validateBinding(binding);
      expect(result.valid).toBe(true);
    });
    
    it('should fail validation for non-existing file', async () => {
      mockSDK.resources.exists.mockResolvedValue(false);
      
      const binding: BindingItem = {
        cardId: '1',
        filePath: '/not-existing.mp4',
        resourceField: 'video_file'
      };
      
      const result = await executor.validateBinding(binding);
      expect(result.valid).toBe(false);
    });
  });
});
```

```typescript
// tests/integration/connector.test.ts

import { describe, it, expect, beforeEach, afterEach } from 'vitest';
import { FileConnector } from '@/core';
import { createMockSDK } from '../fixtures/mockSDK';

describe('FileConnector Integration', () => {
  let connector: FileConnector;
  let mockSDK: any;
  
  beforeEach(async () => {
    mockSDK = createMockSDK();
    connector = new FileConnector({ sdk: mockSDK });
    await connector.initialize();
  });
  
  afterEach(() => {
    connector.destroy();
  });
  
  describe('complete workflow', () => {
    it('should complete manual binding workflow', async () => {
      // 1. 加载卡片
      const cards = await connector.loadCards([
        { id: '1', name: '第01集', type: 'video', path: '', baseCards: [], resourceMode: 'empty', metadata: {} as any }
      ]);
      expect(cards).toHaveLength(1);
      
      // 2. 加载文件
      mockSDK.file.list.mockResolvedValue([
        { name: 'EP01.mp4', path: '/videos/EP01.mp4', size: 100, type: 'video' }
      ]);
      const files = await connector.loadFiles(['/videos']);
      expect(files).toHaveLength(1);
      
      // 3. 设置绑定
      connector.setBindings([
        { cardId: '1', filePath: '/videos/EP01.mp4', resourceField: 'video_file' }
      ]);
      
      // 4. 执行
      mockSDK.card.update.mockResolvedValue({ success: true });
      const result = await connector.execute();
      
      expect(result.success).toBe(1);
      expect(result.failed).toBe(0);
    });
    
    it('should complete auto match workflow', async () => {
      // 准备数据
      const cards = await connector.loadCards([
        { id: '1', name: '第01集', type: 'video', path: '', baseCards: [], resourceMode: 'empty', metadata: {} as any },
        { id: '2', name: '第02集', type: 'video', path: '', baseCards: [], resourceMode: 'empty', metadata: {} as any }
      ]);
      
      mockSDK.file.list.mockResolvedValue([
        { name: 'EP01.mp4', path: '/EP01.mp4', size: 100, type: 'video' },
        { name: 'EP02.mp4', path: '/EP02.mp4', size: 100, type: 'video' }
      ]);
      await connector.loadFiles(['/videos']);
      
      // 自动匹配
      const schemes = await connector.autoMatch('number_sequence');
      
      expect(schemes.length).toBeGreaterThan(0);
      expect(schemes[0].bindings).toHaveLength(2);
      expect(schemes[0].confidence).toBeGreaterThan(80);
    });
  });
});
```

### 3.2 匹配流程集成测试

```typescript
// tests/integration/matching.test.ts

import { describe, it, expect } from 'vitest';
import { SchemeGenerator } from '@/core/matchers';
import type { CardInfo, FileInfo } from '@/types';

describe('Matching Integration', () => {
  const generator = new SchemeGenerator();
  
  describe('scheme generation', () => {
    it('should generate multiple schemes', () => {
      const cards: CardInfo[] = [
        { id: '1', name: '第01集', type: 'video', path: '', baseCards: [], resourceMode: 'empty', metadata: {} as any },
        { id: '2', name: '第02集', type: 'video', path: '', baseCards: [], resourceMode: 'empty', metadata: {} as any }
      ];
      
      const files: FileInfo[] = [
        { name: 'EP01.mp4', path: '/EP01.mp4', size: 100, mimeType: 'video/mp4', type: 'video', extension: 'mp4', modifiedAt: '', createdAt: '' },
        { name: 'EP02.mp4', path: '/EP02.mp4', size: 100, mimeType: 'video/mp4', type: 'video', extension: 'mp4', modifiedAt: '', createdAt: '' }
      ];
      
      const schemes = generator.generateAll(cards, files);
      
      // 应该至少有数字匹配和顺序匹配两种方案
      expect(schemes.length).toBeGreaterThanOrEqual(2);
      
      // 方案应该按置信度排序
      for (let i = 1; i < schemes.length; i++) {
        expect(schemes[i - 1].confidence).toBeGreaterThanOrEqual(schemes[i].confidence);
      }
    });
    
    it('should handle complex naming patterns', () => {
      const cards: CardInfo[] = [
        { id: '1', name: 'Season1-Episode01', type: 'video', path: '', baseCards: [], resourceMode: 'empty', metadata: {} as any },
        { id: '2', name: 'Season1-Episode02', type: 'video', path: '', baseCards: [], resourceMode: 'empty', metadata: {} as any }
      ];
      
      const files: FileInfo[] = [
        { name: 'S01E01.mp4', path: '/S01E01.mp4', size: 100, mimeType: 'video/mp4', type: 'video', extension: 'mp4', modifiedAt: '', createdAt: '' },
        { name: 'S01E02.mp4', path: '/S01E02.mp4', size: 100, mimeType: 'video/mp4', type: 'video', extension: 'mp4', modifiedAt: '', createdAt: '' }
      ];
      
      const schemes = generator.generateAll(cards, files);
      
      // 数字匹配应该能识别
      const numberScheme = schemes.find(s => s.mode === 'number_sequence');
      expect(numberScheme).toBeDefined();
      expect(numberScheme!.bindings).toHaveLength(2);
    });
  });
});
```

```typescript
// tests/e2e/workflows.test.ts

import { test, expect } from '@playwright/test';

test.describe('File Connector Workflows', () => {
  test.beforeEach(async ({ page }) => {
    await page.goto('/');
  });
  
  test('manual binding workflow', async ({ page }) => {
    // 选择卡片
    await page.click('[data-testid="select-cards-btn"]');
    await page.click('[data-testid="card-item-1"]');
    await page.click('[data-testid="confirm-selection"]');
    
    // 添加文件
    await page.click('[data-testid="add-files-btn"]');
    // ... 文件选择
    
    // 建立绑定
    await page.click('[data-testid="binding-row-1"] [data-testid="select-file-btn"]');
    // ... 选择文件
    
    // 预览
    await page.click('[data-testid="preview-btn"]');
    await expect(page.locator('[data-testid="preview-table"]')).toBeVisible();
    
    // 执行

    // 选择卡片和文件
```
