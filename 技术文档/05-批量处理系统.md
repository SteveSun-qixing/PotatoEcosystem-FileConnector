## 绑定执行引擎

### 执行器架构

```typescript
class BindingExecutor {
  private queue: Binding[];
  private processing: boolean = false;
  private progress: ExecutionProgress;
  
  async execute(bindings: Binding[], options: BindingOptions): Promise<ExecutionResult> {
    this.queue = [...bindings];
    this.processing = true;
    
    const results: BindingResult[] = [];
    
    for (let i = 0; i < this.queue.length; i++) {
      if (!this.processing) break; // 支持取消
      
      const binding = this.queue[i];
      
      try {
        await this.processBinding(binding, options);
        results.push({
          binding,
          status: 'success'
        });
      } catch (error) {
        results.push({
          binding,
          status: 'failed',
          error: error.message
        });
      }
      
      // 更新进度
      this.updateProgress(i + 1, this.queue.length);
    }
    
    return {
      total: bindings.length,
      succeeded: results.filter(r => r.status === 'success').length,
      failed: results.filter(r => r.status === 'failed').length,
      results
    };
  }
  
  cancel() {
    this.processing = false;
  }
}
```

### 单个绑定处理

```typescript
async function processBinding(binding: Binding, options: BindingOptions): Promise<void> {
  const { card, file } = binding;
  
  // 1. 读取卡片配置
  const config = await readCardConfig(card.path);
  
  // 2. 处理资源文件
  let resourcePath: string;
  
  if (options.mode === 'copy') {
    // 复制模式：复制文件到卡片目录
    resourcePath = await copyFileToCard(file.path, card.path);
  } else {
    // 引用模式：直接使用文件路径
    resourcePath = file.path;
  }
  
  // 3. 处理路径类型
  if (options.pathType === 'relative') {
    resourcePath = path.relative(card.path, resourcePath);
  }
  
  // 4. 更新配置
  config.resource = {
    type: file.type,
    path: resourcePath,
    size: file.size
  };
  
  // 5. 写回配置
  await writeCardConfig(card.path, config);
}
```

## 文件操作

### 文件复制

```typescript
class FileOperator {
  async copyFile(sourcePath: string, destDir: string): Promise<string> {
    const fileName = path.basename(sourcePath);
    const destPath = path.join(destDir, fileName);
    
    // 检查目标文件是否存在
    if (await fs.pathExists(destPath)) {
      // 处理冲突
      return await this.handleConflict(sourcePath, destPath);
    }
    
    // 复制文件
    await fs.copyFile(sourcePath, destPath);
    
    // 验证复制结果
    await this.verifyFile(sourcePath, destPath);
    
    return destPath;
  }
  
  private async handleConflict(sourcePath: string, destPath: string): Promise<string> {
    // 根据配置决定策略
    switch (this.conflictStrategy) {
      case 'overwrite':
        await fs.copyFile(sourcePath, destPath);
        return destPath;
      
      case 'rename':
        const newPath = this.generateUniqueFileName(destPath);
        await fs.copyFile(sourcePath, newPath);
        return newPath;
      
      case 'skip':
        return destPath;
      
      default:
        throw new Error('File conflict');
    }
  }
  
  private async verifyFile(sourcePath: string, destPath: string): Promise<void> {
    const sourceStats = await fs.stat(sourcePath);
    const destStats = await fs.stat(destPath);
    
    if (sourceStats.size !== destStats.size) {
      throw new Error('File copy verification failed');
    }
  }
}
```

```typescript
class ConfigUpdater {
  async updateCardConfig(cardPath: string, updates: ConfigUpdates): Promise<void> {
    const metadataPath = path.join(cardPath, 'metadata.yaml');
    
    // 读取现有配置
    const content = await fs.readFile(metadataPath, 'utf-8');
    const config = yaml.parse(content);
    
    // 应用更新
    Object.assign(config, updates);
    
    // 写回文件
    const newContent = yaml.stringify(config);
    await fs.writeFile(metadataPath, newContent, 'utf-8');
  }
  
  async batchUpdate(updates: Map<string, ConfigUpdates>): Promise<void> {
    const promises = [];
    
    for (const [cardPath, update] of updates) {
      promises.push(this.updateCardConfig(cardPath, update));
      
      // 限制并发数
      if (promises.length >= 10) {
        await Promise.all(promises);
        promises.length = 0;
      }
    }
    
    if (promises.length > 0) {
      await Promise.all(promises);
    }
  }
}
```

## 进度管理

### 进度跟踪

```typescript
class ProgressTracker {
  private current: number = 0;
  private total: number = 0;
  private startTime: number = 0;
  private callbacks: ProgressCallback[] = [];
  
  start(total: number) {
    this.current = 0;
    this.total = total;
    this.startTime = Date.now();
    this.notifyProgress();
  }
  
  update(increment: number = 1) {
    this.current += increment;
    this.notifyProgress();
  }
  
  onProgress(callback: ProgressCallback) {
    this.callbacks.push(callback);
  }
  
  private notifyProgress() {
    const progress: ProgressInfo = {
      current: this.current,
      total: this.total,
      percentage: (this.current / this.total) * 100,
      elapsed: Date.now() - this.startTime,
      estimated: this.estimateRemaining()
    };
    
    for (const callback of this.callbacks) {
      callback(progress);
    }
  }
  
  private estimateRemaining(): number {
    const elapsed = Date.now() - this.startTime;
    const avgTime = elapsed / this.current;
    return avgTime * (this.total - this.current);
  }
}
```

### 进度显示

```typescript
class ProgressDisplay {
  private element: HTMLElement;
  
  show(progress: ProgressInfo) {
    this.element.innerHTML = `
      <div class="progress-container">
        <div class="progress-bar">
          <div class="progress-fill" style="width: ${progress.percentage}%"></div>
        </div>
        <div class="progress-text">
          ${progress.current} / ${progress.total} (${progress.percentage.toFixed(1)}%)
        </div>
        <div class="progress-time">
          已用时: ${this.formatTime(progress.elapsed)}
          预计剩余: ${this.formatTime(progress.estimated)}
        </div>
      </div>
    `;
  }
  
  private formatTime(ms: number): string {
    const seconds = Math.floor(ms / 1000);
    const minutes = Math.floor(seconds / 60);
    const hours = Math.floor(minutes / 60);
    
    if (hours > 0) {
      return `${hours}小时${minutes % 60}分`;
    } else if (minutes > 0) {
      return `${minutes}分${seconds % 60}秒`;
    } else {
      return `${seconds}秒`;
    }
  }
}
```

```typescript
  PERMISSION_DENIED = 'permission_denied',
```

### 错误恢复

```typescript
class ErrorRecovery {
  async retry(binding: Binding, maxRetries: number = 3): Promise<void> {
    let lastError: Error | null = null;
    
    for (let attempt = 1; attempt <= maxRetries; attempt++) {
      try {
        await processBinding(binding, this.options);
        return; // 成功
      } catch (error) {
        lastError = error;
        
        // 等待后重试
        await this.delay(1000 * attempt);
      }
    }
    
    // 所有重试都失败
    throw lastError;
  }
  
  async recoverFromError(error: BindingError): Promise<boolean> {
    switch (error.type) {
      case BindingErrorType.FILE_NOT_FOUND:
        // 提示用户重新选择文件
        return await this.promptForFile(error.cardId);
      
      case BindingErrorType.PERMISSION_DENIED:
        // 提示用户授予权限
        return await this.requestPermission();
      
      case BindingErrorType.DISK_FULL:
        // 提示用户清理磁盘
        return await this.promptDiskCleanup();
      
      default:
        return false;
    }
  }
}
```

```typescript
class ModeConverter {
  async convertToFullMode(cards: Card[]): Promise<ConversionResult> {
    const results: ConversionResult[] = [];
    
    for (const card of cards) {
      try {
        await this.convertCardToFullMode(card);
        results.push({ card, status: 'success' });
      } catch (error) {
        results.push({ card, status: 'failed', error });
      }
    }
    
    return {
      succeeded: results.filter(r => r.status === 'success').length,
      failed: results.filter(r => r.status === 'failed').length,
      results
    };
  }
  
  private async convertCardToFullMode(card: Card): Promise<void> {
    // 读取配置
    const config = await readCardConfig(card.path);
    
    if (!config.resource || !config.resource.path) {
      throw new Error('No resource to convert');
    }
    
    const resourcePath = config.resource.path;
    
    // 如果已经是相对路径，可能已经是全填充模式
    if (!path.isAbsolute(resourcePath)) {
      const fullPath = path.join(card.path, resourcePath);
      if (await fs.pathExists(fullPath)) {
        return; // 已经是全填充模式
      }
    }
    
    // 复制文件到卡片目录
    const fileName = path.basename(resourcePath);
    const destPath = path.join(card.path, fileName);
    
    await fs.copyFile(resourcePath, destPath);
    
    // 更新配置为相对路径
    config.resource.path = fileName;
    await writeCardConfig(card.path, config);
  }
}
```
