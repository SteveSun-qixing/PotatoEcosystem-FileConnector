```typescript
class TableView {
  private cardList: Card[];
  private fileMap: Map<string, File>;
  private bindings: Map<string, File>;
  
  render() {
    return `
      <table class="binding-table">
        <thead>
          <tr>
            <th>卡片名称</th>
            <th>已绑定文件</th>
            <th>操作</th>
          </tr>
        </thead>
        <tbody>
          ${this.renderRows()}
        </tbody>
      </table>
    `;
  }
  
  renderRows() {
    return this.cardList.map(card => `
      <tr data-card-id="${card.id}">
        <td class="card-name">${card.name}</td>
        <td class="file-cell" ondrop="handleDrop(event)">
          ${this.renderFileInfo(card.id)}
        </td>
        <td class="actions">
          <button onclick="selectFile('${card.id}')">选择</button>
          <button onclick="clearBinding('${card.id}')">清除</button>
        </td>
      </tr>
    `).join('');
  }
}
```

### 拖拽处理

```typescript
class DragDropHandler {
  setupDragDrop() {
    document.querySelectorAll('.file-cell').forEach(cell => {
      cell.addEventListener('dragover', (e) => {
        e.preventDefault();
        e.currentTarget.classList.add('drag-over');
      });
      
      cell.addEventListener('dragleave', (e) => {
        e.currentTarget.classList.remove('drag-over');
      });
      
      cell.addEventListener('drop', (e) => {
        e.preventDefault();
        const cardId = e.currentTarget.closest('tr').dataset.cardId;
        const files = e.dataTransfer.files;
        
        if (files.length > 0) {
          this.handleFileDrop(cardId, files[0]);
        }
      });
    });
  }
  
  handleFileDrop(cardId: string, file: File) {
    // 创建绑定关系
    this.createBinding(cardId, file);
    // 更新界面
    this.updateView();
  }
}
```

## 图形化界面实现

### Canvas 渲染

```typescript
class GraphView {
  private canvas: HTMLCanvasElement;
  private ctx: CanvasRenderingContext2D;
  private cards: CardNode[];
  private files: FileNode[];
  private connections: Connection[];
  
  render() {
    this.clear();
    this.renderCards();
    this.renderFiles();
    this.renderConnections();
  }
  
  renderCards() {
    const startY = 50;
    const spacing = 80;
    
    this.cards.forEach((card, index) => {
      const y = startY + index * spacing;
      this.drawCard(50, y, card);
    });
  }
  
  renderFiles() {
    const startY = 50;
    const spacing = 80;
    
    this.files.forEach((file, index) => {
      const y = startY + index * spacing;
      this.drawFile(350, y, file);
    });
  }
  
  renderConnections() {
    this.connections.forEach(conn => {
      this.drawConnection(conn.from, conn.to);
    });
  }
  
  drawCard(x: number, y: number, card: CardNode) {
    this.ctx.fillStyle = card.bound ? '#4CAF50' : '#CCCCCC';
    this.ctx.fillRect(x, y, 200, 60);
    
    this.ctx.fillStyle = '#000000';
    this.ctx.fillText(card.name, x + 10, y + 35);
  }
  
  drawFile(x: number, y: number, file: FileNode) {
    this.ctx.fillStyle = file.bound ? '#4CAF50' : '#CCCCCC';
    this.ctx.fillRect(x, y, 200, 60);
    
    this.ctx.fillStyle = '#000000';
    this.ctx.fillText(file.name, x + 10, y + 35);
  }
  
  drawConnection(from: Point, to: Point) {
    this.ctx.strokeStyle = '#2196F3';
    this.ctx.lineWidth = 2;
    this.ctx.beginPath();
    this.ctx.moveTo(from.x, from.y);
    
    // 贝塞尔曲线
    const cp1x = from.x + (to.x - from.x) * 0.5;
    const cp1y = from.y;
    const cp2x = from.x + (to.x - from.x) * 0.5;
    const cp2y = to.y;
    
    this.ctx.bezierCurveTo(cp1x, cp1y, cp2x, cp2y, to.x, to.y);
    this.ctx.stroke();
  }
}
```

### 交互处理

```typescript
class GraphInteractionHandler {
  private selectedCard: CardNode | null = null;
  private hoveredNode: Node | null = null;
  
  handleClick(x: number, y: number) {
    const clickedNode = this.findNodeAt(x, y);
    
    if (!clickedNode) {
      this.selectedCard = null;
      return;
    }
    
    if (clickedNode.type === 'card') {
      this.selectedCard = clickedNode;
    } else if (clickedNode.type === 'file' && this.selectedCard) {
      this.createConnection(this.selectedCard, clickedNode);
      this.selectedCard = null;
    }
  }
  
  handleMouseMove(x: number, y: number) {
    const hoveredNode = this.findNodeAt(x, y);
    
    if (hoveredNode !== this.hoveredNode) {
      this.hoveredNode = hoveredNode;
      this.updateCursor();
    }
  }
  
  createConnection(card: CardNode, file: FileNode) {
    this.connections.push({
      cardId: card.id,
      fileId: file.id,
      from: card.position,
      to: file.position
    });
    
    card.bound = true;
    file.bound = true;
    
    this.render();
  }
}
```

## 文件选择对话框

```typescript
class FileSelectDialog {
  async show(cardId: string): Promise<string | null> {
    return new Promise((resolve) => {
      const input = document.createElement('input');
      input.type = 'file';
      
      input.onchange = (e) => {
        const file = (e.target as HTMLInputElement).files?.[0];
        resolve(file ? file.path : null);
      };
      
      input.click();
    });
  }
  
  async showMultiple(): Promise<string[]> {
    return new Promise((resolve) => {
      const input = document.createElement('input');
      input.type = 'file';
      input.multiple = true;
      
      input.onchange = (e) => {
        const files = Array.from((e.target as HTMLInputElement).files || []);
        resolve(files.map(f => f.path));
      };
      
      input.click();
    });
  }
}
```
