# 文件连接器 - 自动识别算法

## 数字序列匹配

### 算法实现

```typescript
class NumberSequenceMatcher {
  match(cards: Card[], files: File[]): Binding[] {
    const cardNumbers = this.extractNumbers(cards);
    const fileNumbers = this.extractNumbers(files);
    
    const bindings: Binding[] = [];
    
    for (const [cardId, numbers] of cardNumbers) {
      for (const [fileId, fileNums] of fileNumbers) {
        if (this.numbersMatch(numbers, fileNums)) {
          bindings.push({
            cardId,
            fileId,
            confidence: 0.95,
            reason: `数字匹配: ${numbers.join(', ')}`
          });
          break;
        }
      }
    }
    
    return bindings;
  }
  
  private extractNumbers(items: Item[]): Map<string, number[]> {
    const result = new Map();
    
    for (const item of items) {
      const numbers = item.name.match(/\d+/g)?.map(n => parseInt(n)) || [];
      result.set(item.id, numbers);
    }
    
    return result;
  }
  
  private numbersMatch(nums1: number[], nums2: number[]): boolean {
    if (nums1.length === 0 || nums2.length === 0) return false;
    
    // 比较第一个数字
    return nums1[0] === nums2[0];
  }
}
```

## 关键词匹配

### 算法实现

```typescript
class KeywordMatcher {
  match(cards: Card[], files: File[]): Binding[] {
    const bindings: Binding[] = [];
    
    for (const card of cards) {
      const cardKeywords = this.extractKeywords(card.name);
      let bestMatch: { file: File; score: number } | null = null;
      
      for (const file of files) {
        const fileKeywords = this.extractKeywords(file.name);
        const score = this.calculateSimilarity(cardKeywords, fileKeywords);
        
        if (score > 0.5 && (!bestMatch || score > bestMatch.score)) {
          bestMatch = { file, score };
        }
      }
      
      if (bestMatch) {
        bindings.push({
          cardId: card.id,
          fileId: bestMatch.file.id,
          confidence: bestMatch.score,
          reason: '关键词匹配'
        });
      }
    }
    
    return bindings;
  }
  
  private extractKeywords(text: string): string[] {
    // 移除特殊字符和数字
    const cleaned = text.replace(/[\d\[\]()（）_-]/g, ' ');
    
    // 分词
    const words = cleaned.toLowerCase().split(/\s+/).filter(w => w.length > 1);
    
    return words;
  }
  
  private calculateSimilarity(keywords1: string[], keywords2: string[]): number {
    const set1 = new Set(keywords1);
    const set2 = new Set(keywords2);
    
    const intersection = new Set([...set1].filter(x => set2.has(x)));
    const union = new Set([...set1, ...set2]);
    
    return intersection.size / union.size;
  }
}
```

## 顺序匹配

```typescript
class OrderMatcher {
  match(cards: Card[], files: File[], options: OrderMatchOptions): Binding[] {
    // 排序卡片
    const sortedCards = this.sortCards(cards, options.cardSortBy);
    
    // 排序文件
    const sortedFiles = this.sortFiles(files, options.fileSortBy);
    
    // 一一对应
    const bindings: Binding[] = [];
    const count = Math.min(sortedCards.length, sortedFiles.length);
    
    for (let i = 0; i < count; i++) {
      bindings.push({
        cardId: sortedCards[i].id,
        fileId: sortedFiles[i].id,
        confidence: 0.7,
        reason: '顺序匹配'
      });
    }
    
    return bindings;
  }
  
  private sortCards(cards: Card[], sortBy: string): Card[] {
    return [...cards].sort((a, b) => {
      switch (sortBy) {
        case 'name':
          return a.name.localeCompare(b.name);
        case 'created':
          return a.createdTime - b.createdTime;
        default:
          return 0;
      }
    });
  }
}
```

## AI 智能匹配

```typescript
class AISmartMatcher {
  async match(cards: Card[], files: File[]): Promise<Binding[]> {
    // 准备数据
    const prompt = this.buildPrompt(cards, files);
    
    // 调用 AI
    const response = await this.callAI(prompt);
    
    // 解析结果
    return this.parseResponse(response);
  }
  
  private buildPrompt(cards: Card[], files: File[]): string {
    return `
你是一个文件匹配助手。请分析以下卡片和文件的对应关系。

卡片列表:
${cards.map(c => `- ${c.id}: ${c.name}`).join('\n')}

文件列表:
${files.map(f => `- ${f.id}: ${f.name}`).join('\n')}

请返回 JSON 格式的匹配结果，格式如下：
{
  "bindings": [
    {
      "cardId": "卡片ID",
      "fileId": "文件ID",
      "confidence": 0.95,
      "reason": "匹配原因"
    }
  ]
}
    `;
  }
  
  private async callAI(prompt: string): Promise<string> {
    // 实现具体的 AI API 调用
    const response = await fetch('https://api.openai.com/v1/chat/completions', {
      method: 'POST',
      headers: {
        'Authorization': `Bearer ${API_KEY}`,
        'Content-Type': 'application/json'
      },
      body: JSON.stringify({
        model: 'gpt-4',
        messages: [{ role: 'user', content: prompt }]
      })
    });
    
    const data = await response.json();
    return data.choices[0].message.content;
  }
  
  private parseResponse(response: string): Binding[] {
    try {
      const data = JSON.parse(response);
      return data.bindings;
    } catch (error) {
      console.error('Failed to parse AI response:', error);
      return [];
    }
  }
}
```

## 自定义规则匹配

```typescript
class CustomRuleMatcher {
  match(cards: Card[], files: File[], rule: MatchingRule): Binding[] {
    const bindings: Binding[] = [];
    
    const cardPattern = new RegExp(rule.cardPattern);
    const filePattern = new RegExp(rule.filePattern);
    
    for (const card of cards) {
      const cardMatch = card.name.match(cardPattern);
      if (!cardMatch) continue;
      
      for (const file of files) {
        const fileMatch = file.name.match(filePattern);
        if (!fileMatch) continue;
        
        if (this.rulesMatch(cardMatch, fileMatch, rule.matchCondition)) {
          bindings.push({
            cardId: card.id,
            fileId: file.id,
            confidence: 0.9,
            reason: `自定义规则: ${rule.name}`
          });
          break;
        }
      }
    }
    
    return bindings;
  }
  
  private rulesMatch(
    cardMatch: RegExpMatchArray,
    fileMatch: RegExpMatchArray,
    condition: string
  ): boolean {
    // 执行匹配条件
    // 例如: "$1 == $1" 表示第一个捕获组相等
    const func = new Function(
      'cardMatch',
      'fileMatch',
      `return ${condition.replace(/\$(\d+)/g, (_, n) => {
        return `(cardMatch[${n}] === fileMatch[${n}])`;
      })}`
    );
    
    return func(cardMatch, fileMatch);
  }
}
```

## 匹配结果评分

```typescript
class MatchingScorer {
  calculateFinalScore(bindings: Binding[]): Binding[] {
    return bindings.map(binding => ({
      ...binding,
      finalScore: this.calculateScore(binding)
    }));
  }
  
  private calculateScore(binding: Binding): number {
    let score = binding.confidence;
    
    // 考虑其他因素
    // 1. 文件类型匹配
    if (this.fileTypeMatches(binding)) {
      score += 0.05;
    }
    
    // 2. 文件大小合理性
    if (this.fileSizeReasonable(binding)) {
      score += 0.02;
    }
    
    return Math.min(score, 1.0);
  }
}
```
