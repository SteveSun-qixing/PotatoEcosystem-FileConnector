# 文件连接器 - 连接模式设计

**版本**: 1.0.0  
**更新时间**: 2026-02-01

---

## 1. 概述

文件连接器提供三种连接模式：

| 模式 | 说明 | 适用场景 |
|------|------|---------|
| 手动连接 | 用户手动建立绑定关系 | 文件数量少、命名无规律 |
| 自动识别 | 根据命名规律自动匹配 | 命名有规律可循 |
| 智能识别 | AI语义分析匹配 | 命名复杂、跨语言 |

```typescript
class ConnectionLine {
  // 计算贝塞尔曲线控制点
  static calculateBezier(
    start: Point,
    end: Point
  ): { start: Point; cp1: Point; cp2: Point; end: Point } {
    const dx = end.x - start.x;
    const cp1 = { x: start.x + dx * 0.4, y: start.y };
    const cp2 = { x: end.x - dx * 0.4, y: end.y };
    
    return { start, cp1, cp2, end };
  }
  
  // 检测点击是否在连线上
  static isPointOnLine(
    point: Point,
    line: ConnectionLine,
    tolerance: number = 5
  ): boolean {
    // 使用细分算法检测
    const samples = this.sampleBezier(line, 20);
    for (const sample of samples) {
      if (this.distance(point, sample) < tolerance) {
        return true;
      }
    }
    return false;
  }
}
```

## 3. 自动识别模式

### 3.1 数字序列匹配

**算法流程**：
```
1. 提取所有数字
   "第01集" → [1]
   "EP01.mp4" → [1]
   
2. 数字标准化
   "01" → 1
   "001" → 1
   
3. 匹配计算
   卡片数字 [1, 2, 3, 4]
   文件数字 [1, 2, 3, 4]
   → 完全匹配
   
4. 生成绑定
   第01集 ↔ EP01.mp4
   第02集 ↔ EP02.mp4
   ...
```

**实现代码**：

```typescript
class NumberSequenceMatcher {
  // 提取数字
  extractNumbers(text: string): number[] {
    const matches = text.match(/\d+/g) || [];
    return matches.map(m => parseInt(m, 10));
  }
  
  // 匹配
  match(cards: CardInfo[], files: FileInfo[]): MatchScheme {
    const cardNumbers = new Map<number, CardInfo>();
    const fileNumbers = new Map<number, FileInfo>();
    
    // 提取卡片数字
    for (const card of cards) {
      const numbers = this.extractNumbers(card.name);
      if (numbers.length > 0) {
        // 默认使用最后一个数字
        cardNumbers.set(numbers[numbers.length - 1], card);
      }
    }
    
    // 提取文件数字
    for (const file of files) {
      const numbers = this.extractNumbers(file.name);
      if (numbers.length > 0) {
        fileNumbers.set(numbers[numbers.length - 1], file);
      }
    }
    
    // 生成匹配
    const bindings: BindingItem[] = [];
    const unmatchedCards: string[] = [];
    const unmatchedFiles: string[] = [];
    
    for (const [num, card] of cardNumbers) {
      const file = fileNumbers.get(num);
      if (file) {
        bindings.push({
          cardId: card.id,
          filePath: file.path,
          resourceField: this.detectResourceField(card, file)
        });
        fileNumbers.delete(num);
      } else {
        unmatchedCards.push(card.id);
      }
    }
    
    for (const [, file] of fileNumbers) {
      unmatchedFiles.push(file.path);
    }
    
    return {
      mode: 'number_sequence',
      confidence: this.calculateConfidence(bindings.length, cards.length),
      bindings,
      unmatchedCards,
      unmatchedFiles
    };
  }
  
  // 计算置信度
  calculateConfidence(matched: number, total: number): number {
    if (total === 0) return 0;
    return Math.round((matched / total) * 100);
  }
}
```

### 3.2 关键词匹配

**算法流程**：
```
1. 分词提取关键词
   "孤独摇滚 第一集" → ["孤独", "摇滚", "第一", "集"]
   "BOCCHI_EP01.mp4" → ["BOCCHI", "EP", "01"]
   
2. 计算相似度
   使用Jaccard相似度或编辑距离
   
3. 找最佳匹配
   每个卡片找相似度最高的文件
   
4. 处理冲突
   如果多个卡片匹配同一文件，保留最高分
```

**实现代码**：

```typescript
class KeywordMatcher {
  // 分词
  tokenize(text: string): string[] {
    // 按非字母数字分割
    const tokens = text
      .toLowerCase()
      .split(/[^a-z0-9\u4e00-\u9fa5]+/)
      .filter(t => t.length > 0);
    
    return tokens;
  }
  
  // Jaccard相似度
  jaccardSimilarity(set1: Set<string>, set2: Set<string>): number {
    const intersection = new Set([...set1].filter(x => set2.has(x)));
    const union = new Set([...set1, ...set2]);
    
    if (union.size === 0) return 0;
    return intersection.size / union.size;
  }
  
  // 匹配
  match(cards: CardInfo[], files: FileInfo[]): MatchScheme {
    const cardTokens = new Map<string, Set<string>>();
    const fileTokens = new Map<string, Set<string>>();
    
    // 分词
    for (const card of cards) {
      cardTokens.set(card.id, new Set(this.tokenize(card.name)));
    }
    for (const file of files) {
      fileTokens.set(file.path, new Set(this.tokenize(file.name)));
    }
    
    // 计算相似度矩阵
    const similarities: Array<{
      cardId: string;
      filePath: string;
      score: number;
    }> = [];
    
    for (const [cardId, cardSet] of cardTokens) {
      for (const [filePath, fileSet] of fileTokens) {
        const score = this.jaccardSimilarity(cardSet, fileSet);
        if (score > 0.1) {  // 阈值
          similarities.push({ cardId, filePath, score });
        }
      }
    }
    
    // 贪心选择最佳匹配
    similarities.sort((a, b) => b.score - a.score);
    
    const usedCards = new Set<string>();
    const usedFiles = new Set<string>();
    const bindings: BindingItem[] = [];
    
    for (const { cardId, filePath, score } of similarities) {
      if (!usedCards.has(cardId) && !usedFiles.has(filePath)) {
        bindings.push({
          cardId,
          filePath,
          resourceField: 'primary_resource'
        });
        usedCards.add(cardId);
        usedFiles.add(filePath);
      }
    }
    
    return {
      mode: 'keyword',
      confidence: this.calculateConfidence(bindings, cards.length),
      bindings,
      unmatchedCards: cards
        .filter(c => !usedCards.has(c.id))
        .map(c => c.id),
      unmatchedFiles: files
        .filter(f => !usedFiles.has(f.path))
        .map(f => f.path)
    };
  }
}
```

### 3.3 文件顺序匹配

**算法流程**：
```
1. 卡片排序
   按名称/创建时间排序
   
2. 文件排序
   按名称/修改时间排序
   
3. 顺序对应
   卡片[0] ↔ 文件[0]
   卡片[1] ↔ 文件[1]
   ...
   
4. 数量校验
   如果数量不一致，生成警告
```

**实现代码**：

```typescript
class FileOrderMatcher {
  // 匹配
  match(
    cards: CardInfo[],
    files: FileInfo[],
    cardSort: 'name' | 'created' = 'name',
    fileSort: 'name' | 'modified' = 'name'
  ): MatchScheme {
    // 排序卡片
    const sortedCards = [...cards].sort((a, b) => {
      if (cardSort === 'name') {
        return a.name.localeCompare(b.name);
      }
      return 0; // 默认不排序
    });
    
    // 排序文件
    const sortedFiles = [...files].sort((a, b) => {
      if (fileSort === 'name') {
        return a.name.localeCompare(b.name);
      }
      return new Date(a.modifiedAt).getTime() - 
             new Date(b.modifiedAt).getTime();
    });
    
    // 顺序匹配
    const bindings: BindingItem[] = [];
    const minLength = Math.min(sortedCards.length, sortedFiles.length);
    
    for (let i = 0; i < minLength; i++) {
      bindings.push({
        cardId: sortedCards[i].id,
        filePath: sortedFiles[i].path,
        resourceField: 'primary_resource'
      });
    }
    
    // 计算置信度（数量一致性）
    const countDiff = Math.abs(cards.length - files.length);
    const confidence = Math.max(0, 100 - countDiff * 10);
    
    return {
      mode: 'file_order',
      confidence,
      bindings,
      unmatchedCards: sortedCards.slice(minLength).map(c => c.id),
      unmatchedFiles: sortedFiles.slice(minLength).map(f => f.path),
      warning: countDiff > 0 
        ? `卡片数量(${cards.length})与文件数量(${files.length})不一致`
        : undefined
    };
  }
}
```

## 4. 智能识别模式

### 4.1 AI服务调用

**请求准备**：

```typescript
class SmartMatcher {
  private aiClient: AIServiceClient;
  
  // 准备请求数据
  prepareRequest(cards: CardInfo[], files: FileInfo[]): AIMatchRequest {
    return {
      cards: cards.map(c => ({
        id: c.id,
        name: c.name,
        type: c.type,
        description: c.description
      })),
      files: files.map(f => ({
        path: f.path,
        name: f.name,
        size: f.size,
        type: this.getFileType(f.name)
      })),
      options: {
        language: 'auto',
        min_confidence: 30
      }
    };
  }
  
  // 执行匹配
  async match(cards: CardInfo[], files: FileInfo[]): Promise<SmartMatchResult> {
    const request = this.prepareRequest(cards, files);
    
    try {
      const response = await this.aiClient.smartMatch(request);
      return this.parseResponse(response);
    } catch (error) {
      // 降级到自动匹配
      console.warn('AI service failed, fallback to auto match');
      return this.fallbackMatch(cards, files);
    }
  }
}
```

### 4.2 置信度分级

**分级规则**：
- **高置信度 (≥80%)**：直接显示为确定匹配
- **中置信度 (50-79%)**：显示为建议匹配，需要用户确认
- **低置信度 (<50%)**：显示为可能匹配，建议手动确认

**实现代码**：

```typescript
interface ClassifiedMatches {
  high: Array<BindingItem & { confidence: number }>;
  medium: Array<BindingItem & { confidence: number }>;
  low: Array<BindingItem & { confidence: number }>;
}

function classifyConfidence(
  matches: Array<BindingItem & { confidence: number }>
): ClassifiedMatches {
  const result: ClassifiedMatches = {
    high: [],
    medium: [],
    low: []
  };
  
  for (const match of matches) {
    if (match.confidence >= 80) {
      result.high.push(match);
    } else if (match.confidence >= 50) {
      result.medium.push(match);
    } else {
      result.low.push(match);
    }
  }
  
  return result;
}
```

class ModeRecommender {
  recommend(cards: CardInfo[], files: FileInfo[]): {
    mode: ConnectorMode;
    reason: string;
    confidence: number;
  } {
    // 检测数字规律
    const numberPattern = this.detectNumberPattern(cards, files);
    if (numberPattern.score > 0.8) {
      return {
        mode: 'auto',
        reason: '检测到文件命名有明显的数字规律',
        confidence: numberPattern.score
      };

    // 文件数量少
    if (cards.length <= 5) {

        mode: 'manual',
        reason: '文件数量较少，推荐手动绑定',
        confidence: 0.9
      };

    return {
      mode: 'smart',
      reason: '文件命名较复杂，推荐使用智能识别',

    };
  }
