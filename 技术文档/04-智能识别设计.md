### OpenAI 集成

```typescript
class OpenAIProvider implements IAIProvider {
  private apiKey: string;
  private model: string = 'gpt-4';
  
  async analyze(cards: Card[], files: File[]): Promise<MatchingResult> {
    const prompt = this.buildPrompt(cards, files);
    
    const response = await fetch('https://api.openai.com/v1/chat/completions', {
      method: 'POST',
      headers: {
        'Authorization': `Bearer ${this.apiKey}`,
        'Content-Type': 'application/json'
      },
      body: JSON.stringify({
        model: this.model,
        messages: [
          {
            role: 'system',
            content: '你是一个专业的文件匹配助手，擅长分析文件名和内容的对应关系。'
          },
          {
            role: 'user',
            content: prompt
          }
        ],
        response_format: { type: 'json_object' }
      })
    });
    
    const data = await response.json();
    return this.parseResponse(data.choices[0].message.content);
  }
  
  private buildPrompt(cards: Card[], files: File[]): string {
    return `
请分析以下卡片和文件的匹配关系：

卡片（共${cards.length}个）:
${cards.map((c, i) => `${i + 1}. ID: ${c.id}, 名称: ${c.name}`).join('\n')}

文件（共${files.length}个）:
${files.map((f, i) => `${i + 1}. ID: ${f.id}, 名称: ${f.name}`).join('\n')}

请返回 JSON 格式，包含：
- bindings: 匹配关系数组
  - cardId: 卡片ID
  - fileId: 文件ID
  - confidence: 置信度（0-1）
  - reason: 匹配理由
- unmatched_cards: 未匹配的卡片ID列表
- unmatched_files: 未匹配的文件ID列表
    `;
  }
}
```

```typescript
class SemanticSimilarity {
  async calculate(text1: string, text2: string): Promise<number> {
    // 使用词嵌入模型计算语义相似度
    const embedding1 = await this.getEmbedding(text1);
    const embedding2 = await this.getEmbedding(text2);
    
    return this.cosineSimilarity(embedding1, embedding2);
  }
  
  private async getEmbedding(text: string): Promise<number[]> {
    // 调用嵌入模型 API
    const response = await fetch('https://api.openai.com/v1/embeddings', {
      method: 'POST',
      headers: {
        'Authorization': `Bearer ${API_KEY}`,
        'Content-Type': 'application/json'
      },
      body: JSON.stringify({
        model: 'text-embedding-ada-002',
        input: text
      })
    });
    
    const data = await response.json();
    return data.data[0].embedding;
  }
  
  private cosineSimilarity(vec1: number[], vec2: number[]): number {
    let dotProduct = 0;
    let norm1 = 0;
    let norm2 = 0;
    
    for (let i = 0; i < vec1.length; i++) {
      dotProduct += vec1[i] * vec2[i];
      norm1 += vec1[i] * vec1[i];
      norm2 += vec2[i] * vec2[i];
    }
    
    return dotProduct / (Math.sqrt(norm1) * Math.sqrt(norm2));
  }
}
```

```typescript
class ConfidenceCalculator {
  calculate(binding: Binding, context: MatchingContext): number {
    let confidence = 0;
    
    // 1. 名称相似度 (40%)
    const nameSimilarity = this.calculateNameSimilarity(
      binding.card.name,
      binding.file.name
    );
    confidence += nameSimilarity * 0.4;
    
    // 2. 类型匹配 (20%)
    if (this.typeMatches(binding.card, binding.file)) {
      confidence += 0.2;
    }
    
    // 3. 上下文一致性 (20%)
    const contextScore = this.evaluateContext(binding, context);
    confidence += contextScore * 0.2;
    
    // 4. AI 建议得分 (20%)
    if (binding.aiScore) {
```

## 批量处理优化

```typescript
class BatchProcessor {
  async processBatch(
    cards: Card[],
    files: File[],
    batchSize: number = 50
  ): Promise<Binding[]> {
    const allBindings: Binding[] = [];
    
    // 分批处理，避免一次请求过大
    for (let i = 0; i < cards.length; i += batchSize) {
      const cardBatch = cards.slice(i, i + batchSize);
      const bindings = await this.aiProvider.analyze(cardBatch, files);
      allBindings.push(...bindings);
      
      // 延迟，避免超过 API 速率限制
      await this.delay(1000);
    }
    
    return allBindings;
  }
  
  private delay(ms: number): Promise<void> {
    return new Promise(resolve => setTimeout(resolve, ms));
  }
}
```

## 结果后处理

```typescript
class ResultPostProcessor {
  process(bindings: Binding[]): ProcessedBinding[] {
    // 1. 去重
    const uniqueBindings = this.removeDuplicates(bindings);
    
    // 2. 冲突解决
    const resolvedBindings = this.resolveConflicts(uniqueBindings);
    
    // 3. 置信度分级
    return this.classifyByConfidence(resolvedBindings);
  }
  
  private removeDuplicates(bindings: Binding[]): Binding[] {
    const seen = new Set<string>();
    return bindings.filter(b => {
      const key = `${b.cardId}-${b.fileId}`;
      if (seen.has(key)) return false;
      seen.add(key);
      return true;
    });
  }
  
  private resolveConflicts(bindings: Binding[]): Binding[] {
    // 一个卡片只能绑定一个文件
    // 如果有多个候选，选择置信度最高的
    const cardMap = new Map<string, Binding>();
    
    for (const binding of bindings) {
      const existing = cardMap.get(binding.cardId);
      if (!existing || binding.confidence > existing.confidence) {
        cardMap.set(binding.cardId, binding);
      }
    }
    
    return Array.from(cardMap.values());
  }
  
  private classifyByConfidence(bindings: Binding[]): ProcessedBinding[] {
    return bindings.map(b => ({
      ...b,
      level: this.getConfidenceLevel(b.confidence)
    }));
  }
  
  private getConfidenceLevel(confidence: number): 'high' | 'medium' | 'low' {
    if (confidence >= 0.8) return 'high';
    if (confidence >= 0.5) return 'medium';
    return 'low';
  }
}
```
